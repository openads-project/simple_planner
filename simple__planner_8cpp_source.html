310SimplePlannerNode::PlannerState SimplePlannerNode::determinePlannerState(const rclcpp::Time& stamp) {
311 // ego data missing -> no publish
312 if (!ego_data_init_) {
313 setHealth(diagnostic_msgs::msg::DiagnosticStatus::STALE, "No ego data received yet",
314 {{"PlannerState", plannerStateToString(PlannerState::NoPublish)}});
315 return PlannerState::NoPublish;
316 }
317
318 // ego data outdated -> no publish
319 if (isMessageOutdated(ego_data_.header, ego_data_timeout_, stamp)) {
320 ego_data_init_ = false;
321 std::string msg =
322 "EgoData is older than " + std::to_string(ego_data_timeout_) + " seconds. Skip publishing until fresh ego data arrives.";
323 RCLCPP_DEBUG(this->get_logger(), "%s", msg.c_str());
324 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg,
325 {{"PlannerState", plannerStateToString(PlannerState::NoPublish)}});
326 return PlannerState::NoPublish;
327 }
328
329 // no route received and no ongoing safe stop -> no publish
330 if (!route_init_ && !safe_stop_distance_.has_value()) {
331 setHealth(diagnostic_msgs::msg::DiagnosticStatus::STALE, "No route received and no ongoing safe stop",
332 {{"PlannerState", plannerStateToString(PlannerState::NoPublish)}});
333 return PlannerState::NoPublish;
334 }
335
336 // route outdated and vehicle in standstill -> standstill
337 if (route_init_ && isMessageOutdated(route_.header, route_timeout_, stamp)) {
338 if (perception_msgs::object_access::getStandstill(ego_data_)) {
339 route_init_ = false;
340 safe_stop_distance_.reset();
341 latest_path_.points.clear();
342 std::string msg = "Route is older than " + std::to_string(route_timeout_) +
343 " seconds and ego vehicle is stationary. Publishing standstill trajectory.";
344 RCLCPP_DEBUG(this->get_logger(), "%s", msg.c_str());
345 setHealth(diagnostic_msgs::msg::DiagnosticStatus::OK, msg,
346 {{"PlannerState", plannerStateToString(PlannerState::Standstill)}});
347 return PlannerState::Standstill;
348 }
349 route_init_ = false;
350
351 // route outdated and vehicle still moving -> safe stop
352 std::string msg = "Route is older than " + std::to_string(route_timeout_) +
353 " seconds but ego vehicle is still moving. Executing safe stop trajectory.";
354 RCLCPP_DEBUG(this->get_logger(), "%s", msg.c_str());
355 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg,
356 {{"PlannerState", plannerStateToString(PlannerState::SafeStop)}});
357 return PlannerState::SafeStop;
358 }
359
360 // route received, but empty -> standstill
361 if (route_init_ && route_.route_elements.empty()) {
362 route_init_ = false;
363 safe_stop_distance_.reset();
364 latest_path_.points.clear();
365 std::string msg = "Received route has no route elements. Publishing standstill trajectory.";
366 RCLCPP_DEBUG(this->get_logger(), "%s", msg.c_str());
367 setHealth(diagnostic_msgs::msg::DiagnosticStatus::OK, msg,
368 {{"PlannerState", plannerStateToString(PlannerState::Standstill)}});
369 return PlannerState::Standstill;
370 }
371
372 // no fresh route, but safe stop already started -> safe stop
373 if (!route_init_ && safe_stop_distance_.has_value()) {
374 if (perception_msgs::object_access::getStandstill(ego_data_)) {
375 safe_stop_distance_.reset();
376 latest_path_.points.clear();
377 std::string msg = "Safe stop finished. Ego vehicle is considered stationary. Publishing standstill trajectory.";
378 RCLCPP_DEBUG(this->get_logger(), "%s", msg.c_str());
379 setHealth(diagnostic_msgs::msg::DiagnosticStatus::OK, msg,
380 {{"PlannerState", plannerStateToString(PlannerState::Standstill)}});
381 return PlannerState::Standstill;
382 }
383 std::string msg = "No fresh route available, but safe stop already started. Executing safe stop trajectory.";
384 RCLCPP_DEBUG(this->get_logger(), "%s", msg.c_str());
385 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg,
386 {{"PlannerState", plannerStateToString(PlannerState::SafeStop)}});
387 return PlannerState::SafeStop;
388 }
389
390 // fresh ego data and valid route available -> follow route
391 setHealth(diagnostic_msgs::msg::DiagnosticStatus::OK, "Input information up to date. Following route.",
392 {{"PlannerState", plannerStateToString(PlannerState::FollowRoute)}});
393 return PlannerState::FollowRoute;
394}

412trajectory_planning_msgs::msg::Trajectory SimplePlannerNode::createTrajectory(PlannerState state, const rclcpp::Time& stamp) {
413 rclcpp::Time begin = rclcpp::Clock(RCL_SYSTEM_TIME).now();
414
415 trajectory_planning_msgs::msg::Trajectory tra;
416 tra.header.stamp = stamp;
417 tra.header.frame_id = trajectory_frame_id_;
418
419 if (state != PlannerState::FollowRoute) {
420 resetObjectState(tra.header);
421 }
422
423 switch (state) {
424 case PlannerState::Standstill:
425 tra = buildStandstillTrajectory(tra.header);
426 break;
427 case PlannerState::SafeStop:
428 if (!safe_stop_distance_.has_value()) {
429 tra = buildTrajectoryFromSimplePath(buildSafeStopPath(tra.header));
430 } else {
431 RCLCPP_DEBUG(this->get_logger(), "Executing safe stop.");
432 tra = buildTrajectoryFromSimplePath(transformPath(latest_path_, tra.header));
433 }
434 break;
435 case PlannerState::FollowRoute: {
436 safe_stop_distance_.reset();
437 FollowRoutePlan route_plan = buildRoutePlan(tra.header);
438 tra = buildTrajectoryFromSimplePath(route_plan.path);
439 break;
440 }
441 case PlannerState::NoPublish:
442 throw std::runtime_error("createTrajectory called for non-publish state");
443 }
444
445 rclcpp::Time end = rclcpp::Clock(RCL_SYSTEM_TIME).now();
446 RCLCPP_DEBUG(this->get_logger(), "Trajectory creation took %f ms", (end - begin).seconds() * 1e3);
447 return tra;
448}

460trajectory_planning_msgs::msg::Trajectory SimplePlannerNode::buildTrajectoryFromSimplePath(const SimplePath& path) {
461 SimplePath usable_path = path;
462 trimPathBehindEgo(usable_path);
463
464 if (usable_path.points.empty()) {
465 safe_stop_distance_.reset();
466 latest_path_.points.clear();
467 std::string msg = "No usable forward path remains. Publishing standstill trajectory.";
468 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
469 health_.key_value_pairs.insert_or_assign("PlannerState", plannerStateToString(PlannerState::Standstill));
470 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
471 return buildStandstillTrajectory(path.header);
472 }
473
474 latest_path_ = usable_path;
475 std::vector<SimplePathPoint> path_points = latest_path_.points;
476 const auto n_states_size = static_cast<size_t>(n_states_);
477 if (n_states_size < path_points.size()) {
478 path_points.resize(n_states_size);
479 }
480
481 int type_id = trajectory_planning_msgs::REFERENCE::TYPE_ID;
482 trajectory_planning_msgs::msg::Trajectory tra;
483 trajectory_planning_msgs::trajectory_access::initializeTrajectory(tra, type_id, n_states_);
484 tra.header = usable_path.header;
485
486 for (int i = 0; i < n_states_; i++) {
487 const size_t idx = static_cast<size_t>(i) < path_points.size() ? static_cast<size_t>(i) : path_points.size() - 1;
488 trajectory_planning_msgs::trajectory_access::setT(tra, dt_ * i, i);
489 trajectory_planning_msgs::trajectory_access::setX(tra, path_points[idx].position.x(), i);
490 trajectory_planning_msgs::trajectory_access::setY(tra, path_points[idx].position.y(), i);
491 trajectory_planning_msgs::trajectory_access::setV(tra, path_points[idx].v, i);
492 RCLCPP_DEBUG(this->get_logger(), "Debug: i: %d, t: %f, x: %f, y: %f, v: %f, s: %f", i, dt_ * i,
493 path_points[idx].position.x(), path_points[idx].position.y(), path_points[idx].v, path_points[idx].s);
494 }
495
496 trajectory_planning_msgs::trajectory_access::setStandstill(tra, false);
497 RCLCPP_DEBUG(this->get_logger(), "Standstill = %d", tra.standstill);
498 return tra;
499}

501SimplePath SimplePlannerNode::buildSafeStopPath(const std_msgs::msg::Header& target_header) {
502 double current_velocity = perception_msgs::object_access::getVelocityMagnitude(ego_data_);
503 safe_stop_distance_ = -0.5 * std::pow(current_velocity, 2) / a_max_decel_;
504
505 SimplePath safe_stop_path = transformPath(latest_path_, target_header);
506 trimPathBehindEgo(safe_stop_path);
507 if (safe_stop_path.points.empty()) {
508 RCLCPP_WARN(
509 this->get_logger(),
510 "No latest path available. Initialize safe stop along ego heading. Current velocity: %f m/s, safe stop distance: %f m",
511 current_velocity, *safe_stop_distance_);
512 latest_path_ = transformPath(calculateSafeStopAlongEgoHeading(ego_data_, *safe_stop_distance_, target_header), target_header);
513 } else {
514 RCLCPP_WARN(this->get_logger(), "Initialize safe stop along latest path. Current velocity: %f m/s, safe stop distance: %f m",
515 current_velocity, *safe_stop_distance_);
516 latest_path_ = calculateSafeStopAlongRoute(safe_stop_path, *safe_stop_distance_);
517 }
518 return latest_path_;

521SimplePlannerNode::FollowRoutePlan SimplePlannerNode::buildRoutePlan(const std_msgs::msg::Header& target_header) {
522 RCLCPP_DEBUG(this->get_logger(), "Default case: route is up to date, creating path from route.");
523
524 geometry_msgs::msg::TransformStamped tf;
525 try {
526 tf = tf2_buffer_->lookupTransform(target_header.frame_id, target_header.stamp, route_.header.frame_id, route_.header.stamp,
527 fixed_over_time_frame_id_, rclcpp::Duration::from_seconds(1.0));
528 } catch (tf2::TransformException& ex) {
529 std::string msg = "Transformation is not available: " + std::string(ex.what()) + ".";
530 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
531 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
532 }
533
534 route_planning_msgs::msg::Route tf_route;
535 tf2::doTransform(route_, tf_route, tf);
536 FollowRoutePlan route_plan;
537 route_plan.path.header = tf_route.header;
538
539 std::map<uint64_t, uint64_t> lane_change_indices_map;
540 appendRoutePoints(tf_route, route_plan, lane_change_indices_map);
541
542 std::vector<SimplePathPoint> merged_points = mergeLaneChangeSegments(tf_route, route_plan.path.points, lane_change_indices_map);
543 recalculateS(merged_points);
544 route_plan.path.points = resamplePath(merged_points, route_plan.stop_at_end, route_plan.offset_to_stop_line);
545 applyObjectConstraints(target_header, merged_points, route_plan);
546 if (trigger_turn_signals_) {
547 applyIndicatorRequest(route_plan.suggested_turn_signal);
548 health_.key_value_pairs.insert_or_assign("SuggestedTurnSignal", turnSignalToString(route_plan.suggested_turn_signal));
549 }
550 if (route_plan.stop_at_end) health_.key_value_pairs.insert({"ReasonToStop", route_plan.reason_to_stop});
551 return route_plan;
552}

554void SimplePlannerNode::appendRoutePoints(const route_planning_msgs::msg::Route& tf_route,
555 FollowRoutePlan& route_plan,
556 std::map<uint64_t, uint64_t>& lane_change_indices_map) {
557 double t_total = 0.0;
558 RCLCPP_DEBUG(this->get_logger(), "Number of remaining route elements: %zu",
559 tf_route.destination_route_element_idx - tf_route.current_route_element_idx);
560 health_.key_value_pairs.insert(
561 {"RemainingRouteElements", std::to_string(tf_route.destination_route_element_idx - tf_route.current_route_element_idx)});
562 for (size_t j = tf_route.current_route_element_idx; j < tf_route.destination_route_element_idx; ++j) {
563 const auto& route_element = tf_route.route_elements[j];
564 if (!route_element.is_enriched) {
565 RCLCPP_DEBUG(this->get_logger(), "Route element %zu is not enriched. Skipping.", j);
566 continue;
567 }
568
569 const auto& suggested_lane = route_planning_msgs::route_access::getSuggestedLaneElement(route_element);
570 SimplePathPoint simple_path_point;
571 simple_path_point.position =
572 Eigen::Vector2d(suggested_lane.reference_pose.position.x, suggested_lane.reference_pose.position.y);
573 simple_path_point.s = route_element.s;
574 simple_path_point.v = v_ref_;
575 if (v_ref_ < 0.0) {
576 simple_path_point.v = suggested_lane.speed_limit / 3.6;
577 }
578
579 if (!route_plan.path.points.empty()) {
580 double v_average = (route_plan.path.points.back().v + simple_path_point.v) / 2.0;
581 double dt = 0.0;
582 if (v_average != 0.0) {
583 dt = (simple_path_point.s - route_plan.path.points.back().s) / v_average;
584 }
585 if (dt <= 0.0) {
586 std::string msg = "Negative time difference " + std::to_string(dt) +
587 " between points at s=" + std::to_string(route_plan.path.points.back().s) +
588 " and s=" + std::to_string(simple_path_point.s) + ". Could lead to unexpected behavior.";
589 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
590 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
591 }
592 t_total += dt;
593 }
594
595 if (j == tf_route.current_route_element_idx) {
596 route_plan.suggested_turn_signal = suggested_lane.suggested_turn_signal;
597 }
598
599 if (route_element.will_change_suggested_lane &&
600 !tryRegisterLaneChange(tf_route, j, lane_change_indices_map, route_plan.suggested_turn_signal)) {
601 break;
602 }
603
604 if (consider_traffic_lights_) {
605 updateForTrafficLights(tf_route, j, suggested_lane, simple_path_point, t_total, route_plan.stop_at_end,
606 route_plan.offset_to_stop_line);
607 if (route_plan.stop_at_end) route_plan.reason_to_stop = "Traffic light indicates stop";
608 }
609
610 route_plan.path.points.push_back(simple_path_point);
611
612 if (j == tf_route.destination_route_element_idx - 1) {
613 SimplePathPoint destination_point;
614 destination_point.position = Eigen::Vector2d(tf_route.destination.x, tf_route.destination.y);
615 destination_point.s = simple_path_point.s + (destination_point.position - simple_path_point.position).norm();
616 destination_point.v = simple_path_point.v;
617 route_plan.path.points.push_back(destination_point);
618 route_plan.reason_to_stop = "Reaching end of route";
619 route_plan.stop_at_end = true;
620 }
621 if (route_plan.stop_at_end || t_total >= 2.0 * trajectory_horizon_) {
622 break;
623 }
624 }

627bool SimplePlannerNode::tryRegisterLaneChange(const route_planning_msgs::msg::Route& tf_route,
628 size_t route_element_idx,
629 std::map<uint64_t, uint64_t>& lane_change_indices_map,
630 uint8_t& suggested_turn_signal) {
631 if (route_element_idx + 1 >= tf_route.route_elements.size()) {
632 std::string msg = "Route element " + std::to_string(route_element_idx) + " is the last element. Cannot change lane.";
633 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
634 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
635 return false;
636 }
637
638 const size_t i_end = route_element_idx + 1;
639 const auto& route_element = tf_route.route_elements[route_element_idx];
640 size_t current_lane_idx = route_element.suggested_lane_idx;
641 int lane_change_direction = 0;
642 try {
643 lane_change_direction =
644 route_planning_msgs::route_access::getLaneChangeDirection(route_element, tf_route.route_elements[route_element_idx + 1]);
645 } catch (const std::exception& ex) {
646 RCLCPP_WARN(this->get_logger(), "Could not determine lane change direction at route element %zu: %s", route_element_idx,
647 ex.what());
648 return false;
649 }
650 if (lane_change_direction < 0) {
651 suggested_turn_signal = route_planning_msgs::msg::LaneElement::SUGGESTED_TURN_SIGNAL_LEFT;
652 } else if (lane_change_direction > 0) {
653 suggested_turn_signal = route_planning_msgs::msg::LaneElement::SUGGESTED_TURN_SIGNAL_RIGHT;
654 }
655
656 double ego_velocity = perception_msgs::object_access::getVelocityMagnitude(ego_data_);
657 double lane_change_distance =
658 std::max(lane_change_min_distance_factor_ * ego_data_.length, lane_change_distance_factor_ * ego_velocity);
659 RCLCPP_DEBUG(this->get_logger(), "Lane change direction: %d, lane change distance: %f", lane_change_direction,
660 lane_change_distance);
661 if (lane_change_direction == 0) {
662 RCLCPP_WARN(this->get_logger(),
663 "Route element %zu is marked as lane change, but suggested lane does not change. Ignoring lane change marker.",
664 route_element_idx);
665 return true;
666 }
667
668 double ds = 0.0;
669 size_t i_start = route_element_idx;
670 while (ds < lane_change_distance && i_start > 0) {
671 if (auto result = route_planning_msgs::route_access::getPrecedingLaneElementIdx(current_lane_idx,
672 tf_route.route_elements[i_start - 1])) {
673 current_lane_idx = *result;
674 } else {
675 std::string msg =
676 "No preceding lane element found for route element " + std::to_string(i_start) + ". Cannot extend lane change.";
677 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
678 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
679 break;
680 }
681 if (i_start >= 2 && tf_route.route_elements[i_start - 2].will_change_suggested_lane) {
682 std::string msg = "Found previous lane change in route element " + std::to_string(i_start - 2) +
683 ". Could not extend lane change over this element.";
684 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
685 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
686 break;
687 }
688 if (!route_planning_msgs::route_access::hasAdjacentLane(tf_route.route_elements[i_start - 1], current_lane_idx,
689 lane_change_direction)) {
690 std::string msg = "No adjacent lane found for route element " + std::to_string(i_start - 1) + ".";
691 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
692 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
693 break;
694 }
695 ds += std::abs(tf_route.route_elements[i_start].s - tf_route.route_elements[i_start - 1].s);
696 i_start--;
697 }
698
699 if (!tf_route.route_elements[i_start].is_enriched || !tf_route.route_elements[i_end].is_enriched) {
700 std::string msg =
701 "Not enough enriched route elements (" + std::to_string(i_start) + ", " + std::to_string(i_end) + ") for lane change.";
702 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
703 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
704 return false;
705 }
706
707 lane_change_indices_map[route_element_idx] = i_start;
708 return true;
709}

711void SimplePlannerNode::updateForTrafficLights(const route_planning_msgs::msg::Route& tf_route,
712 size_t route_element_idx,
713 const route_planning_msgs::msg::LaneElement& suggested_lane,
714 const SimplePathPoint& simple_path_point,
715 double t_total,
716 bool& stop_at_end,
717 double& offset_to_stop_line) {
718 const auto& route_element = tf_route.route_elements[route_element_idx];
719 const auto& reg_elems =
720 route_planning_msgs::route_access::getRegulatoryElementsOfLaneElement(suggested_lane, route_element.regulatory_elements);
721 for (size_t k = 0; k < reg_elems.size(); ++k) {
722 if (reg_elems[k].type != route_planning_msgs::msg::RegulatoryElement::TYPE_TRAFFIC_LIGHT) {
723 continue;
724 }
725 if (reg_elems[k].meta_value == route_planning_msgs::msg::RegulatoryElement::META_VALUE_MOVEMENT_ALLOWED &&
726 !consider_future_states_) {
727 continue;
728 }
729
730 offset_to_stop_line =
731 offset_to_stop_line_ + ego_data_.length / 2.0 + ego_data_.state.reference_point.translation_to_geometric_center.x;
732 double dt_offset_to_stop_line = 0.0;
733 if (simple_path_point.v != 0.0) {
734 dt_offset_to_stop_line = offset_to_stop_line / simple_path_point.v;
735 }
736 if (dt_offset_to_stop_line <= 0.0) {
737 std::string msg = "Negative time difference 'dt_offset_to_stop_line' (" + std::to_string(dt_offset_to_stop_line) +
738 " s) for traffic light at route element " + std::to_string(route_element_idx) +
739 ". Could lead to unexpected behavior.";
740 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
741 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
742 }
743
744 if (reg_elems[k].has_validity_stamp && consider_future_states_) {
745 double validity_duration =
746 rclcpp::Time(reg_elems[k].validity_stamp).seconds() - rclcpp::Time(route_.header.stamp).seconds();
747 if (validity_duration < (t_total - dt_offset_to_stop_line)) {
748 if (reg_elems[k].meta_value == route_planning_msgs::msg::RegulatoryElement::META_VALUE_MOVEMENT_ALLOWED) {
749 stop_at_end = true;
750 } else {
751 continue;
752 }
753 } else {
754 if (reg_elems[k].meta_value == route_planning_msgs::msg::RegulatoryElement::META_VALUE_MOVEMENT_ALLOWED) {
755 continue;
756 } else {
757 stop_at_end = true;
758 }
759 }
760 } else {
761 stop_at_end = true;
762 }
763
764 double distance_to_stop_point = tf_route.route_elements[route_element_idx].s -
765 tf_route.route_elements[tf_route.current_route_element_idx].s - offset_to_stop_line;
766 double v_ego = perception_msgs::object_access::getVelocityMagnitude(ego_data_);
767 double min_distance_to_stop = -0.5 * std::pow(v_ego, 2) / a_max_decel_;
768 if (distance_to_stop_point < 0.0 && std::abs(distance_to_stop_point) > ignore_stop_line_threshold_) {
769 std::string msg =
770 "Traffic light stop point is behind ego vehicle (distance to stop point: " + std::to_string(distance_to_stop_point) +
771 " m). Ignoring traffic light.";
772 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
773 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
774 stop_at_end = false;
775 } else if ((distance_to_stop_point < min_distance_to_stop) && stop_at_end) {
776 std::string msg =
777 "Traffic light requires stop, but distance to stop point is smaller than minimum distance to stop. Ignoring traffic "
778 "light.";
779 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
780 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
781 stop_at_end = false;
782 }
783 }
784}

786std::vector<SimplePathPoint> SimplePlannerNode::mergeLaneChangeSegments(
787 const route_planning_msgs::msg::Route& tf_route,
788 const std::vector<SimplePathPoint>& route_points,
789 const std::map<uint64_t, uint64_t>& lane_change_indices_map) {
790 size_t current = 0;
791 std::vector<SimplePathPoint> merged_points;
792 for (const auto& lane_change_indices : lane_change_indices_map) {
793 uint64_t lane_change_idx_route = lane_change_indices.first;
794 uint64_t start_idx_route = lane_change_indices.second;
795 size_t start_idx =
796 start_idx_route > tf_route.current_route_element_idx ? start_idx_route - tf_route.current_route_element_idx : 0;
797 size_t end_idx = lane_change_idx_route + 1 > tf_route.current_route_element_idx
798 ? lane_change_idx_route + 1 - tf_route.current_route_element_idx
799 : 0;
800
801 if (start_idx < current || current > route_points.size()) {
802 RCLCPP_WARN(this->get_logger(), "Skipping overlapping lane change window (%zu, %zu), current path index: %zu.", start_idx,
803 end_idx, current);
804 continue;
805 }
806
807 start_idx = std::min(start_idx, route_points.size());
808 const auto current_offset = static_cast<std::vector<SimplePathPoint>::difference_type>(current);
809 const auto start_offset = static_cast<std::vector<SimplePathPoint>::difference_type>(start_idx);
810 merged_points.insert(merged_points.end(), route_points.begin() + current_offset, route_points.begin() + start_offset);
811 std::vector<SimplePathPoint> lane_change_points = generateLaneChangePath(start_idx_route, lane_change_idx_route, tf_route);
812 merged_points.insert(merged_points.end(), lane_change_points.begin(), lane_change_points.end());
813 current = std::min(end_idx + 1, route_points.size());
814 }
815 if (current <= route_points.size()) {
816 const auto current_offset = static_cast<std::vector<SimplePathPoint>::difference_type>(current);
817 merged_points.insert(merged_points.end(), route_points.begin() + current_offset, route_points.end());
818 }
819 return merged_points;
820}

860SimplePath SimplePlannerNode::calculateSafeStopAlongRoute(const SimplePath& path, const double safe_stop_distance) {
861 SimplePath safe_stop_path = path;
862 double start_s = safe_stop_path.points[0].s; // start s value of path
863 for (size_t i = 0; i < safe_stop_path.points.size(); i++) {
864 if (safe_stop_path.points[i].s > start_s + safe_stop_distance) {
865 // remove all points after the point where the safe stop distance is reached
866 const auto erase_offset = static_cast<std::vector<SimplePathPoint>::difference_type>(i);
867 safe_stop_path.points.erase(safe_stop_path.points.begin() + erase_offset, safe_stop_path.points.end());
868 break;
869 }
870 }
871 safe_stop_path.points = resamplePath(safe_stop_path.points, true, 0.0);
872 return safe_stop_path;
873}

875SimplePath SimplePlannerNode::calculateSafeStopAlongEgoHeading(const perception_msgs::msg::EgoData& ego_data,
876 const double safe_stop_distance,
877 const std_msgs::msg::Header& target_header) {
878 SimplePath safe_stop_path;
879 safe_stop_path.header = ego_data.header;
880
881 double v_ego = perception_msgs::object_access::getVelocityMagnitude(ego_data);
882 geometry_msgs::msg::Point point = perception_msgs::object_access::getPosition(ego_data);
883 double yaw = perception_msgs::object_access::getYaw(ego_data);
884 Eigen::Vector2d start_position(point.x, point.y);
885 Eigen::Vector2d heading(std::cos(yaw), std::sin(yaw));
886
887 RCLCPP_WARN(this->get_logger(), "Initializing minimal safe stop along ego heading. Frame: %s, yaw: %f rad",
888 safe_stop_path.header.frame_id.c_str(), yaw);
889 (void)target_header;
890
891 SimplePathPoint start_point(start_position, 0.0, v_ego);
892 safe_stop_path.points.push_back(start_point);
893 SimplePathPoint mid_point(start_position + heading * (safe_stop_distance / 2.0), safe_stop_distance / 2.0, v_ego);
894 safe_stop_path.points.push_back(mid_point);
895 SimplePathPoint end_point(start_position + heading * safe_stop_distance, safe_stop_distance, 0.0);
896 safe_stop_path.points.push_back(end_point);
897
898 safe_stop_path.points = resamplePath(safe_stop_path.points, true, 0.0);
899 return safe_stop_path;
900}

902std::vector<SimplePathPoint> SimplePlannerNode::generateLaneChangePath(size_t start_idx,
903 size_t turn_idx,
904 const route_planning_msgs::msg::Route& route) {
905 std::vector<SimplePathPoint> lane_change_path;
906 const size_t end_idx = turn_idx + 1; // lane change should end at the next route element
907 if (start_idx >= end_idx || end_idx >= route.route_elements.size()) {
908 std::string msg = "Invalid lane change indices: start_idx=" + std::to_string(start_idx) +
909 ", end_idx=" + std::to_string(end_idx) + ". Cannot generate lane change path.";
910 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
911 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
912 return lane_change_path;
913 }
914
915 int lane_change_direction = route_planning_msgs::route_access::getLaneChangeDirection(route.route_elements[turn_idx],
916 route.route_elements[turn_idx + 1]);
917
918 // Interpolate between the two elements
919 for (size_t i = start_idx; i <= end_idx; ++i) {
920 if (i < route.current_route_element_idx) continue;
921
922 const auto& route_element = route.route_elements[i];
923 const auto& suggested_lane = route_planning_msgs::route_access::getSuggestedLaneElement(route_element);
924 if (i > turn_idx) lane_change_direction = 0; // -> i == turn_idx + 1 == end_idx
925 const auto& adjacent_lane = route_planning_msgs::route_access::getAdjacentLane(
926 route_element, route.route_elements[i].suggested_lane_idx, lane_change_direction);
927 Eigen::Vector2d suggested_lane_pos(suggested_lane.reference_pose.position.x, suggested_lane.reference_pose.position.y);

932 lane_change_path.push_back(
933 SimplePathPoint(interpolated_pos, route_element.s, suggested_lane.speed_limit / 3.6)); // convert km/h to m/s
934 }

949std::vector<SimplePathPoint> SimplePlannerNode::resamplePath(const std::vector<SimplePathPoint>& path,
950 bool stop_at_end,
951 double offset_to_stop_line,
952 const double* speed_cap) {
953 rclcpp::Time begin = rclcpp::Clock(RCL_SYSTEM_TIME).now();
954 if (path.empty()) {
955 std::string msg = "Route is empty. No resampling possible.";
956 RCLCPP_WARN(this->get_logger(), "%s", msg.c_str());
957 setHealth(diagnostic_msgs::msg::DiagnosticStatus::WARN, msg, health_.key_value_pairs);
958 return path;
959 }
960
961 std::vector<SimplePathPoint> resampled_path;
962 double s = path[0].s;
963
964 tk::spline x_spline, y_spline;
965 if (interpolation_type_ == InterpolationType::SPLINE && path.size() > 2) {
966 std::vector<double> s_vector, x_vector, y_vector;
967 for (size_t j = 0; j < path.size(); ++j) {
968 s_vector.push_back(path[j].s);
969 x_vector.push_back(path[j].position.x());
970 y_vector.push_back(path[j].position.y());
971 }
972 x_spline.set_points(s_vector, x_vector);
973 y_spline.set_points(s_vector, y_vector);
974 }
975
976 while (s < path.back().s) {
977 // find index of segment in route (not required if using linearInterpolation from utils)
978 int idx = -1;
979 for (size_t j = 0; j < path.size() - 1; ++j) {
980 if (s >= path[j].s && s <= path[j + 1].s) {
981 idx = static_cast<int>(j);
982 break;
983 }
984 }
985
986 double v = v_ref_; // option 1: use predefined constant velocity
987 if (v_ref_ < 0.0 && idx >= 0) { // option 2: use velocity from route if predefined velocity is negative
988 v = path[idx].v + (path[idx + 1].v - path[idx].v) / (path[idx + 1].s - path[idx].s) * (s - path[idx].s);
989 }
990 if (speed_cap != nullptr) {
991 v = std::min(v, std::max(*speed_cap, 0.0));
992 }
993
994 double distance_to_stop = -0.5 * std::pow(v, 2) / a_decel_ + offset_to_stop_line;
995 distance_to_stop = std::max(distance_to_stop, 0.0);
996 double brake_point = path.back().s - distance_to_stop;
997 double ds = v * dt_; // case 1: constant velocity
998 if (s + ds > brake_point && stop_at_end) {
999 if (s < brake_point) { // special case: braking point is between two states
1000 double ds_1 = brake_point - s; // distance with constant velocity to braking point
1001 double dt_1 = ds_1 / v; // time with constant velocity to braking point
1002 double dt_2 = dt_ - dt_1; // remaining time with deceleration
1003 double ds_2 = std::max(0.5 * a_decel_ * std::pow(dt_2, 2) + v * dt_2, 0.0);
1004 ds = ds_1 + ds_2;
1005 } else {
1006 v = std::sqrt(std::max(std::pow(v, 2) + 2 * a_decel_ * (s - brake_point), 0.0)); // case 2: deceleration (v(s))
1007 ds = 0.5 * a_decel_ * std::pow(dt_, 2) + v * dt_; // case 2: deceleration
1008 if (ds < 0.0) ds = path.back().s - s; // only add rest of route instead of driving backwards
1009 }
1010 }
1011
1012 // interpolate point at s
1013 SimplePathPoint simple_path_point;
1014 if (path.size() == 1) {
1015 simple_path_point.position = path[0].position;
1016 } else if (interpolation_type_ == InterpolationType::SPLINE && path.size() > 2) { // spline interpolation
1017 simple_path_point.position.x() = x_spline(s);
1018 simple_path_point.position.y() = y_spline(s);
1019 } else if (
1020 (interpolation_type_ == InterpolationType::SPLINE && path.size() <= 2) ||
1021 interpolation_type_ ==
1022 InterpolationType::
1023 LINEAR) { // linear interpolation // TODO: could be improved by using our linearInterpolation function -> no need for idx anymore
1024 simple_path_point.position.x() = path[idx].position.x() + (path[idx + 1].position.x() - path[idx].position.x()) /
1025 (path[idx + 1].s - path[idx].s) * (s - path[idx].s);
1026 simple_path_point.position.y() = path[idx].position.y() + (path[idx + 1].position.y() - path[idx].position.y()) /
1027 (path[idx + 1].s - path[idx].s) * (s - path[idx].s);
1028 } else { // unsupported interpolation type
1029 RCLCPP_ERROR(this->get_logger(), "Unsupported interpolation type value %d", interpolation_type_);
1030 throw std::runtime_error("Unsupported interpolation type value");
1031 }
1032 simple_path_point.s = s;
1033 simple_path_point.v = v;
1034 resampled_path.push_back(simple_path_point);
1035
1036 // increment s and v for next iteration
1037 if (v <= 1e-6 && ds <= 1e-6) break;
1038 s = s + ds;
1039 if (s == path.back().s && v == 0.0) break; // stop at end of route
1040 }
1041
1042 rclcpp::Time end = rclcpp::Clock(RCL_SYSTEM_TIME).now();
1043 RCLCPP_DEBUG(this->get_logger(), "Resampling route took %f ms", (end - begin).seconds() * 1e3);
1044
1045 if (stop_at_end) {
1046 SimplePathPoint stop_point = path.back();
1047 if (!resampled_path.empty() && (offset_to_stop_line > 0.0 || (speed_cap != nullptr && *speed_cap <= 1e-6))) {
1048 stop_point = resampled_path.back();
1049 }
1050 stop_point.v = 0.0;
1051 if (resampled_path.empty() || resampled_path.back().s < stop_point.s || resampled_path.back().v != 0.0) {
1052 resampled_path.push_back(stop_point);
1053 }
1054 }
1055
1056 return resampled_path;
1057}
